"""
THE HOUSE ON MERCER LANE — Geometry Builder
PSX-style, Loop One palette
All dimensions from ART_DIRECTION.md (1 unit = 1 meter)

import numpy as np
import math

# =============================================
# PALETTE — Loop One: "Just Another Night"
# =============================================
COLORS = {
    'primary_warm':   (0.769, 0.584, 0.416),   # #C4956A wood paneling, furniture
    'kitchen_light':  (0.910, 0.784, 0.510),   # #E8C882 overhead fluorescent
    'shadow_base':    (0.173, 0.122, 0.078),   # #2C1F14 dark corners

    'accent_rust':    (0.545, 0.271, 0.075),   # #8B4513 Ray's bottles, worn wood
    'ceiling_off':    (0.890, 0.878, 0.851),   # slightly yellowed ceiling

    'basement_bulb':  (0.831, 0.682, 0.376),   # #D4AE60 warm isolated bulb
    'vcr_green':      (0.000, 1.000, 0.255),   # #00FF41 VCR clock
    'leo_red':        (0.800, 0.133, 0.133),   # #CC2222 Leo's shirt
    'couch_fabric':   (0.502, 0.400, 0.302),   # worn couch upholstery
    'glass_window':   (0.700, 0.800, 0.900),   # window glass tint
    'trim_white':     (0.920, 0.910, 0.890),   # door trim, window trim
    'rug_dark':       (0.310, 0.220, 0.150),   # living room area rug
    'countertop':     (0.690, 0.670, 0.620),   # kitchen counter
}

class Mesh:
    """Simple mesh container: vertices, faces, normals, UVs, material"""
    def __init__(self, name, material='wall_paint'):
        self.name = name
        self.material = material
        self.vertices = []   # list of (x, y, z)
        self.faces = []      # list of (i0, i1, i2) or (i0,i1,i2,i3)
        self.normals = []    # per-face normals
        self.uvs = []        # per-face UV coords list of [(u,v),(u,v),...]
        self.color = COLORS.get(material, COLORS['wall_paint'])

    def add_quad(self, v0, v1, v2, v3, uv_scale=1.0, flip_normal=False):
        """Add a quad face (as two triangles), auto-compute normal"""
        base = len(self.vertices)
        self.vertices.extend([v0, v1, v2, v3])
        # Normal from first triangle
        a = np.array(v1) - np.array(v0)
        b = np.array(v2) - np.array(v0)
        n = np.cross(a, b)
        length = np.linalg.norm(n)
        if length > 1e-6:
            n = n / length
        if flip_normal:
            n = -n
        n = tuple(n)
        # Triangle 1: v0, v1, v2
        self.faces.append((base, base+1, base+2))
        self.normals.append(n)
        self.uvs.append([(0,0),(uv_scale,0),(uv_scale,uv_scale)])
        # Triangle 2: v0, v2, v3
        self.faces.append((base, base+2, base+3))
        self.normals.append(n)
        self.uvs.append([(0,0),(uv_scale,uv_scale),(0,uv_scale)])

    def add_box(self, min_pt, max_pt, mat_override=None):
        """Add a solid box from min to max point, all 6 faces"""
        x0,y0,z0 = min_pt
        x1,y1,z1 = max_pt
        if mat_override:
            old_mat = self.material
            self.material = mat_override
        # Bottom (z=z0, looking down)
        self.add_quad((x0,y0,z0),(x1,y0,z0),(x1,y1,z0),(x0,y1,z0), flip_normal=True)
        # Top (z=z1, looking up)
        self.add_quad((x0,y0,z1),(x0,y1,z1),(x1,y1,z1),(x1,y0,z1))
        # Front (y=y0)
        self.add_quad((x0,y0,z0),(x0,y0,z1),(x1,y0,z1),(x1,y0,z0))
        # Back (y=y1)
        self.add_quad((x0,y1,z0),(x1,y1,z0),(x1,y1,z1),(x0,y1,z1))
        # Left (x=x0)
        self.add_quad((x0,y0,z0),(x0,y1,z0),(x0,y1,z1),(x0,y0,z1))
        # Right (x=x1)
        self.add_quad((x1,y0,z0),(x1,y0,z1),(x1,y1,z1),(x1,y1,z0))


def make_room_shell(name, x, y, z, w, d, h, floor_mat, wall_mat, ceil_mat, has_floor=True, has_ceiling=True):
    """Build a room shell as separate meshes for floor/walls/ceiling"""
    meshes = []

    if has_floor:
        floor = Mesh(f"{name}_floor", floor_mat)
        floor.add_quad((x,y,z),(x+w,y,z),(x+w,y+d,z),(x,y+d,z))
        meshes.append(floor)

    if has_ceiling:
        ceil = Mesh(f"{name}_ceiling", ceil_mat)
        ceil.add_quad((x,y,z+h),(x,y+d,z+h),(x+w,y+d,z+h),(x+w,y,z+h))
        meshes.append(ceil)

    # Walls: -Y, +Y, -X, +X (skipping door/window openings for now)
    wall_ny = Mesh(f"{name}_wall_front", wall_mat)
    wall_ny.add_quad((x,y,z),(x,y,z+h),(x+w,y,z+h),(x+w,y,z))
    meshes.append(wall_ny)

    wall_py = Mesh(f"{name}_wall_back", wall_mat)
    wall_py.add_quad((x,y+d,z),(x+w,y+d,z),(x+w,y+d,z+h),(x,y+d,z+h))
    meshes.append(wall_py)

    wall_nx = Mesh(f"{name}_wall_left", wall_mat)
    wall_nx.add_quad((x,y,z),(x,y+d,z),(x,y+d,z+h),(x,y,z+h))
    meshes.append(wall_nx)

    wall_px = Mesh(f"{name}_wall_right", wall_mat)
    wall_px.add_quad((x+w,y,z),(x+w,y,z+h),(x+w,y+d,z+h),(x+w,y+d,z))
    meshes.append(wall_px)

    return meshes


def make_door(name, x, y, z, facing='x', width=0.9, height=2.0, color='door_wood'):
    """Simple PSX door - flat panel with frame"""
    meshes = []
    door = Mesh(f"{name}_door", color)
    thickness = 0.05
    if facing == 'x':
        door.add_box((x, y, z), (x+thickness, y+width, z+height))
    else:
        door.add_box((x, y, z), (x+width, y+thickness, z+height))
    meshes.append(door)
    return meshes


def make_staircase(name, x, y, z, steps=7, step_w=1.0, step_h=0.18, step_d=0.25):
    """7-step basement staircase"""
    meshes = []
    stair = Mesh(f"{name}_stairs", 'stair_wood')
    for i in range(steps):
        sx = x
        sy = y + i * step_d
        sz = z + i * step_h
        # Step tread
        stair.add_quad(
            (sx, sy, sz),
            (sx+step_w, sy, sz),
            (sx+step_w, sy+step_d, sz),
            (sx, sy+step_d, sz)
        )
        # Step riser
        stair.add_quad(
            (sx, sy, sz-step_h if i>0 else z-0.01),
            (sx, sy, sz),
            (sx+step_w, sy, sz),
            (sx+step_w, sy, sz-step_h if i>0 else z-0.01)
        )
    meshes.append(stair)
    return meshes


def make_window(name, x, y, z, w=1.2, h=1.0, facing='y'):
